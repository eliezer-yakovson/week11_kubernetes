# Commands to be used
# Step 1: Clone the repository
git clone .....
# Step 2: Build the Docker image
docker build -t streamlit_app:latest .
# Step 3: Run the Docker container locally (for testing)
docker run -p 8080:8080 streamlit_app:latest
# Step 4: Start Minikube
minikube start

# Step 5: Apply the Pod configuration
kubectl apply -f pod.yaml
